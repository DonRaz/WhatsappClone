// FILE _______________ App.tsx _____________

import { useEffect, useState, useRef } from 'react';
import { AudioVisualizer } from '../components/AudioVisualizer';
import Progress from '../components/Progress';
import { LanguageSelector } from '../components/LanguageSelector';
import { 
  WorkerMessage, 
  AudioState, 
  ModelState, 
  AudioConfig 
} from './types';

const IS_WEBGPU_AVAILABLE = !!navigator.gpu;

const AUDIO_CONFIG: AudioConfig = {
  sampleRate: 16_000,
  maxAudioLength: 30,
  maxSamples: 16_000 * 30
};

const App: React.FC = () => {
  // Worker reference
  const worker = useRef<Worker | null>(null);
  const recorderRef = useRef<MediaRecorder | null>(null);

  // Model state
  const [modelState, setModelState] = useState<ModelState>({
    status: null,
    loadingMessage: '',
    progressItems: [],
    text: '',
    tps: null,
    language: 'en'
  });

  // Audio state
  const [audioState, setAudioState] = useState<AudioState>({
    recording: false,
    isProcessing: false,
    chunks: [],
    stream: null
  });

  const audioContextRef = useRef<AudioContext | null>(null);

  useEffect(() => {
    if (!worker.current) {
      worker.current = new Worker(new URL('./workers/speech.worker.ts', import.meta.url), {
        type: 'module'
      });
    }

    const onMessageReceived = (e: MessageEvent<WorkerMessage>) => {
      switch (e.data.status) {
        case 'loading':
          setModelState(prev => ({
            ...prev,
            status: 'loading',
            loadingMessage: e.data.data
          }));
          break;

        case 'initiate':
          setModelState(prev => ({
            ...prev,
            progressItems: [...prev.progressItems, e.data]
          }));
          break;

        case 'progress':
          setModelState(prev => ({
            ...prev,
            progressItems: prev.progressItems.map(item => 
              item.file === e.data.file ? { ...item, ...e.data } : item
            )
          }));
          break;

        case 'done':
          setModelState(prev => ({
            ...prev,
            progressItems: prev.progressItems.filter(
              item => item.file !== e.data.file
            )
          }));
          break;

        case 'ready':
          setModelState(prev => ({ ...prev, status: 'ready' }));
          recorderRef.current?.start();
          break;

        case 'start':
          setAudioState(prev => ({ ...prev, isProcessing: true }));
          recorderRef.current?.requestData();
          break;

        case 'update':
          const { tps } = e.data;
          setModelState(prev => ({ ...prev, tps }));
          break;

        case 'complete':
          setAudioState(prev => ({ ...prev, isProcessing: false }));
          setModelState(prev => ({ ...prev, text: e.data.output }));
          break;
      }
    };

    worker.current.addEventListener('message', onMessageReceived);

    return () => {
      worker.current?.removeEventListener('message', onMessageReceived);
    };
  }, []);

  useEffect(() => {
    if (recorderRef.current) return;

    const initializeAudio = async () => {
      if (navigator.mediaDevices.getUserMedia) {
        try {
          const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
          setAudioState(prev => ({ ...prev, stream }));

          recorderRef.current = new MediaRecorder(stream);
          audioContextRef.current = new AudioContext({ 
            sampleRate: AUDIO_CONFIG.sampleRate 
          });

          recorderRef.current.onstart = () => {
            setAudioState(prev => ({
              ...prev,
              recording: true,
              chunks: []
            }));
          };

          recorderRef.current.ondataavailable = (e: BlobEvent) => {
            if (e.data.size > 0) {
              setAudioState(prev => ({
                ...prev,
                chunks: [...prev.chunks, e.data]
              }));
            } else {
              setTimeout(() => {
                recorderRef.current?.requestData();
              }, 25);
            }
          };

          recorderRef.current.onstop = () => {
            setAudioState(prev => ({ ...prev, recording: false }));
          };

        } catch (err) {
          console.error("Media access error:", err);
        }
      }
    };

    initializeAudio();

    return () => {
      recorderRef.current?.stop();
      recorderRef.current = null;
    };
  }, []);

  useEffect(() => {
    const processAudioChunks = async () => {
      if (!recorderRef.current) return;
      if (!audioState.recording) return;
      if (audioState.isProcessing) return;
      if (modelState.status !== 'ready') return;
      if (audioState.chunks.length === 0) {
        recorderRef.current.requestData();
        return;
      }

      const blob = new Blob(audioState.chunks, { 
        type: recorderRef.current.mimeType 
      });

      const fileReader = new FileReader();

      fileReader.onloadend = async () => {
        const arrayBuffer = fileReader.result as ArrayBuffer;
        const decoded = await audioContextRef.current!.decodeAudioData(arrayBuffer);
        let audio = decoded.getChannelData(0);
        
        if (audio.length > AUDIO_CONFIG.maxSamples) {
          audio = audio.slice(-AUDIO_CONFIG.maxSamples);
        }

        worker.current?.postMessage({ 
          type: 'generate', 
          data: { 
            audio, 
            language: modelState.language 
          } 
        });
      };

      fileReader.readAsArrayBuffer(blob);
    };

    processAudioChunks();
  }, [
    modelState.status,
    audioState.recording,
    audioState.isProcessing,
    audioState.chunks,
    modelState.language
  ]);

  if (!IS_WEBGPU_AVAILABLE) {
    return (
      <div className="fixed w-screen h-screen bg-black z-10 bg-opacity-[92%] text-white text-2xl font-semibold flex justify-center items-center text-center">
        WebGPU is not supported<br />by this browser :&#40;
      </div>
    );
  }

  return (
    <div className="flex flex-col h-screen mx-auto justify-end text-gray-800 dark:text-gray-200 bg-white dark:bg-gray-900">
      <div className="h-full overflow-auto scrollbar-thin flex justify-center items-center flex-col relative">
        {/* Logo and Title Section */}
        <div className="flex flex-col items-center mb-1 max-w-[400px] text-center">
          <img src="logo.png" width="50%" height="auto" className="block" alt="Logo" />
          <h1 className="text-4xl font-bold mb-1">Whisper WebGPU</h1>
          <h2 className="text-xl font-semibold">Real-time in-browser speech recognition</h2>
        </div>

        <div className="flex flex-col items-center px-4">
          {/* Initial Load Section */}
          {modelState.status === null && (
            <>
              <p className="max-w-[480px] mb-4">
                <br />
                You are about to load <a href="https://huggingface.co/onnx-community/whisper-base" target="_blank" rel="noreferrer" className="font-medium underline">whisper-base</a>,
                a 73 million parameter speech recognition model that is optimized for inference on the web. Once downloaded, the model (~200&nbsp;MB) will be cached and reused when you revisit the page.<br />
                <br />
                Everything runs directly in your browser using <a href="https://huggingface.co/docs/transformers.js" target="_blank" rel="noreferrer" className="underline">🤗&nbsp;Transformers.js</a> and ONNX Runtime Web,
                meaning no data is sent to a server. You can even disconnect from the internet after the model has loaded!
              </p>

              <button
                className="border px-4 py-2 rounded-lg bg-blue-400 text-white hover:bg-blue-500 disabled:bg-blue-100 disabled:cursor-not-allowed select-none"
                onClick={() => {
                  worker.current?.postMessage({ type: 'load' });
                  setModelState(prev => ({ ...prev, status: 'loading' }));
                }}
                disabled={modelState.status !== null}
              >
                Load model
              </button>
            </>
          )}

          {/* Audio Visualizer and Output Section */}
          <div className="w-[500px] p-2">
            <AudioVisualizer 
              className="w-full rounded-lg" 
              stream={audioState.stream} 
            />
            {modelState.status === 'ready' && (
              <div className="relative">
                <p className="w-full h-[80px] overflow-y-auto overflow-wrap-anywhere border rounded-lg p-2">
                  {modelState.text}
                </p>
                {modelState.tps && (
                  <span className="absolute bottom-0 right-0 px-1">
                    {modelState.tps.toFixed(2)} tok/s
                  </span>
                )}
              </div>
            )}
          </div>

          {/* Language Selector and Reset Section */}
          {modelState.status === 'ready' && (
            <div className='relative w-full flex justify-center'>
              <LanguageSelector 
                language={modelState.language} 
                setLanguage={(newLang: string) => {
                  recorderRef.current?.stop();
                  setModelState(prev => ({ ...prev, language: newLang }));
                  recorderRef.current?.start();
                }} 
              />
              <button 
                className="border rounded-lg px-2 absolute right-2" 
                onClick={() => {
                  recorderRef.current?.stop();
                  recorderRef.current?.start();
                }}
              >
                Reset
              </button>
            </div>
          )}

          {/* Loading Progress Section */}
          {modelState.status === 'loading' && (
            <div className="w-full max-w-[500px] text-left mx-auto p-4">
              <p className="text-center">{modelState.loadingMessage}</p>
              {modelState.progressItems.map((item, i) => (
                <Progress 
                  key={i} 
                  text={item.file} 
                  percentage={item.progress} 
                  total={item.total} 
                />
              ))}
            </div>
          )}
        </div>
      </div>
    </div>
  );
};

export default App;

// FILE _______________ speech.worker.ts _____________

import {
    AutoTokenizer,
    AutoProcessor,
    WhisperForConditionalGeneration,
    TextStreamer,
    full,
    PreTrainedTokenizer,
    PreTrainedModel
} from '@huggingface/transformers';
import { 
    WorkerMessage, 
    WorkerRequest, 
    GenerateParams,
    ASRPipelineConfig 
} from './types';

const MAX_NEW_TOKENS = 64;

const PIPELINE_CONFIG: ASRPipelineConfig = {
    modelId: 'onnx-community/whisper-base',
    maxNewTokens: MAX_NEW_TOKENS,
    device: 'webgpu',
    dtype: {
        encoder_model: 'fp32',
        decoder_model_merged: 'q4'
    }
};

interface StreamerConfig {
    skip_prompt: boolean;
    skip_special_tokens: boolean;
    callback_function: (output: string) => void;
}

class AutomaticSpeechRecognitionPipeline {
    private static instance: AutomaticSpeechRecognitionPipeline;
    private static tokenizer: PreTrainedTokenizer | null = null;
    private static processor: PreTrainedModel | null = null;
    private static model: WhisperForConditionalGeneration | null = null;

    private constructor() {}

    static async getInstance(
        progress_callback: ((progress: WorkerMessage) => void) | null = null
    ): Promise<[PreTrainedTokenizer, PreTrainedModel, WhisperForConditionalGeneration]> {
        if (!this.tokenizer) {
            this.tokenizer = await AutoTokenizer.from_pretrained(
                PIPELINE_CONFIG.modelId,
                { progress_callback }
            );
        }

        if (!this.processor) {
            this.processor = await AutoProcessor.from_pretrained(
                PIPELINE_CONFIG.modelId,
                { progress_callback }
            );
        }

        if (!this.model) {
            this.model = await WhisperForConditionalGeneration.from_pretrained(
                PIPELINE_CONFIG.modelId,
                {
                    dtype: PIPELINE_CONFIG.dtype,
                    device: PIPELINE_CONFIG.device,
                    progress_callback,
                }
            );
        }

        return [this.tokenizer, this.processor, this.model];
    }
}

let processing = false;

async function generate({ audio, language }: GenerateParams): Promise<void> {
    if (processing) return;
    processing = true;

    // Notify start of processing
    self.postMessage({ status: 'start' } as WorkerMessage);

    try {
        const [tokenizer, processor, model] = await AutomaticSpeechRecognitionPipeline.getInstance();

        let startTime: number | undefined;
        let numTokens = 0;

        const callback_function = (output: string) => {
            startTime ??= performance.now();

            let tps: number | undefined;
            if (numTokens++ > 0) {
                tps = (numTokens / (performance.now() - startTime)) * 1000;
            }

            self.postMessage({
                status: 'update',
                output,
                tps,
                numTokens,
            } as WorkerMessage);
        };

        const streamerConfig: StreamerConfig = {
            skip_prompt: true,
            skip_special_tokens: true,
            callback_function,
        };

        const streamer = new TextStreamer(tokenizer, streamerConfig);
        const inputs = await processor(audio);

        const outputs = await model.generate({
            ...inputs,
            max_new_tokens: MAX_NEW_TOKENS,
            language,
            streamer,
        });

        const outputText = tokenizer.batch_decode(outputs, { 
            skip_special_tokens: true 
        });

        self.postMessage({
            status: 'complete',
            output: outputText,
        } as WorkerMessage);

    } catch (error) {
        console.error('Generation error:', error);
        self.postMessage({
            status: 'complete',
            output: 'Error during transcription.',
        } as WorkerMessage);
    } finally {
        processing = false;
    }
}

async function load(): Promise<void> {
    try {
        self.postMessage({
            status: 'loading',
            data: 'Loading model...'
        } as WorkerMessage);

        const [tokenizer, processor, model] = await AutomaticSpeechRecognitionPipeline.getInstance(
            (progress: WorkerMessage) => {
                self.postMessage(progress);
            }
        );

        self.postMessage({
            status: 'loading',
            data: 'Compiling shaders and warming up model...'
        } as WorkerMessage);

        // Warm up model with dummy input
        await model.generate({
            input_features: full([1, 80, 3000], 0.0),
            max_new_tokens: 1,
        });

        self.postMessage({ status: 'ready' } as WorkerMessage);

    } catch (error) {
        console.error('Loading error:', error);
        self.postMessage({
            status: 'loading',
            data: 'Error loading model.'
        } as WorkerMessage);
    }
}

// Web Worker message handler
self.addEventListener('message', async (e: MessageEvent<WorkerRequest>) => {
    const { type, data } = e.data;

    switch (type) {
        case 'load':
            await load();
            break;

        case 'generate':
            if (data) {
                await generate(data);
            }
            break;
    }
});

// src/types/worker.types.ts

export type WorkerStatus = 
  | 'loading' 
  | 'initiate' 
  | 'progress' 
  | 'done' 
  | 'ready' 
  | 'start' 
  | 'update' 
  | 'complete';

export interface WorkerMessage {
  status: WorkerStatus;
  data?: any;
  file?: string;
  progress?: number;
  total?: number;
  output?: string;
  tps?: number;
}

export interface GenerateParams {
  audio: Float32Array;
  language: string;
}

export type WorkerMessageType = 'load' | 'generate';

export interface WorkerRequest {
  type: WorkerMessageType;
  data?: GenerateParams;
}

// src/types/audio.types.ts

export interface AudioState {
  recording: boolean;
  isProcessing: boolean;
  chunks: Blob[];
  stream: MediaStream | null;
}

export interface AudioConfig {
  sampleRate: number;
  maxAudioLength: number;
  maxSamples: number;
}

// src/types/model.types.ts

export interface ProgressItem {
  file: string;
  progress: number;
  total: number;
}

export interface WorkerProgressMessage extends WorkerMessage {
  file: string;
  progress: number;
  total: number;
}

export interface ModelState {
  status: WorkerStatus | null;
  loadingMessage: string;
  progressItems: ProgressItem[];
  text: string;
  tps: number | null;
  language: string;
}

export interface ASRPipelineConfig {
  modelId: string;
  maxNewTokens: number;
  device: 'webgpu';
  dtype: {
    encoder_model: 'fp32' | 'fp16';
    decoder_model_merged: 'q4' | 'fp32';
  };
}
